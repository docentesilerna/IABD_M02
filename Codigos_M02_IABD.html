2.1.1Aprendizaje supervisado

Pág. 20‑21Python

Estructura básica de un problema supervisado: carga el dataset Iris, mezcla y divide los datos a mano, entrena una regresión logística y compara predicciones con valores reales.

import numpy as np
from sklearn.datasets import load_iris
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score

np.random.seed(42)  # Para reproducibilidad, fijamos la semilla aleatoria

# 1. Cargar el dataset
# load_iris() devuelve un objeto con atributos:
#   - data: las características (features) de cada muestra
#   - target: la etiqueta de cada muestra
#   - feature_names: nombres de las características
iris = load_iris()
X = iris.data    # Matriz de características, forma (150, 4)
y = iris.target  # Vector de etiquetas, forma (150,)
print("Dimensiones de X:", X.shape)
print("Dimensiones de y:", y.shape)

# 2. Mezclar y dividir los datos en entrenamiento (train) y prueba (test)
test_ratio = 0.2  # 20 % de los datos para test, 80 % para entrenamiento
num_samples = X.shape[0]
num_test = int(num_samples * test_ratio)
num_train = num_samples - num_test

# Mezclamos los índices aleatoriamente
indices = np.arange(num_samples)
np.random.shuffle(indices)

# Dividimos los datos
train_indices = indices[:num_train]
test_indices = indices[num_train:]
X_train, y_train = X[train_indices], y[train_indices]
X_test, y_test = X[test_indices], y[test_indices]
print("Número de muestras de entrenamiento:", X_train.shape[0])
print("Número de muestras de prueba:", X_test.shape[0])

# 3. Crear y entrenar el modelo (regresión logística, un modelo supervisado clásico)
model = LogisticRegression(max_iter=1000)
model.fit(X_train, y_train)

# 4. Hacer predicciones y evaluar
y_pred = model.predict(X_test)
print("Exactitud (accuracy):", accuracy_score(y_test, y_pred))

# 5. Analizar las predicciones
for i in range(len(y_test)):
    print(f"Muestra {i}: Real={y_test[i]}, Predicha={y_pred[i]}")

import matplotlib.pyplot as plt
import pandas as pd
import plotly.express as px
import seaborn as sns
from sklearn.datasets import load_wine
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

pca.fit(scaled_data)

# Varianza explicada acumulada según el número de componentes

    "components": list(range(1, len(df_cluster.columns) + 1)),
    "explainedVariance": pca.explained_variance_ratio_,
})
plot_data["explainedVariance"] = plot_data["explainedVariance"].cumsum()
print(plot_data)

sns.lineplot(data=plot_data, x="components", y="explainedVariance")
plt.title("Varianza explicada acumulada por componentes principales")
plt.gcf().set_size_inches(12, 8)  # Tamaño de la figura
plt.show()

# Scatterplot 3D de las tres primeras componentes
pca = PCA(n_components=3)
pca_data = pca.fit_transform(scaled_data)
pca_data = pd.DataFrame(pca_data,
                        columns=["component_1", "component_2", "component_3"])
pca_data["target"] = wine.target.astype(str)

fig = px.scatter_3d(pca_data, x="component_1", y="component_2",
                    z="component_3", color="target", height=600)

4.3Algoritmos de reglas de asociación

Pág. 63Python

Ejemplo 1. Obtiene conjuntos frecuentes y reglas de asociación con Apriori sobre unas transacciones de cesta de la compra.

# ¡Es una librería aparte de scikit-learn!
# Para instalarla: pip install mlxtend
import pandas as pd
from mlxtend.frequent_patterns import apriori, association_rules

# Creamos un conjunto de transacciones simplificado
dataset = [
    ["pan", "leche", "mantequilla"],
    ["pan", "leche"],
    ["pan", "mantequilla"],
    ["leche", "galletas"],
    ["pan", "galletas", "mantequilla"],
]

# Convertimos las transacciones en una matriz binaria de presencia/ausencia
items = sorted({item for transaction in dataset for item in transaction})
df = pd.DataFrame([[item in t for item in items] for t in dataset],
                  columns=items)

# Aplicamos Apriori con un soporte mínimo
frequent_itemsets = apriori(df, min_support=0.4, use_colnames=True)

# Generamos reglas de asociación con confianza y lift
rules = association_rules(frequent_itemsets, metric="confidence",
                          min_threshold=0.5)

print("Reglas generadas:")
print(rules[["antecedents", "consequents", "support", "confidence", "lift"]])

Ejemplo 2. Imputación condicional: rellena los valores con la media, la mediana o la moda de su subgrupo.

import numpy as np
import pandas as pd

# DataFrame de ejemplo con valores ausentes
df = pd.DataFrame({
    "ocupacion": ["ingeniero", "ingeniero", "docente",
                  "docente", "docente", "ingeniero"],
    "edad": [34, 41, 29, np.nan, 52, 38],
    "renta": [42000, np.nan, 28000, 31000, np.nan, 45000],
})

columna_a_imputar = "renta"       # Columna que se desea imputar
columna_subgrupo = "ocupacion"    # Columna que define los subgrupos
grupos = df.groupby(columna_subgrupo)[columna_a_imputar]

# Opción A: media según el subgrupo
df_media = df.copy()
df_media[columna_a_imputar] = df_media[columna_a_imputar].fillna(
    grupos.transform("mean"))

# Opción B: mediana según el subgrupo
df_mediana = df.copy()
df_mediana[columna_a_imputar] = df_mediana[columna_a_imputar].fillna(
    grupos.transform("median"))

# Opción C: moda según el subgrupo
df_moda = df.copy()
df_moda[columna_a_imputar] = df_moda[columna_a_imputar].fillna(
    grupos.transform(lambda x: x.mode()[0]))

from sklearn.feature_selection import SelectKBest, chi2
from sklearn.datasets import load_breast_cancer

Ejemplo 2. Método de envoltura: eliminación recursiva de características (RFE) con una regresión logística.

from sklearn.datasets import load_breast_cancer
from sklearn.feature_selection import RFE
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler

X, y = load_breast_cancer(return_X_y=True)
X_scaled = StandardScaler().fit_transform(X)

modelo = LogisticRegression(max_iter=500)
selector = RFE(modelo, n_features_to_select=8)
selector.fit(X_scaled, y)
print("Características seleccionadas:", selector.support_)

from scipy.stats import randint
from sklearn.datasets import load_breast_cancer

7.2.4Ejemplo básico de red neuronal

Pág. 100‑103Python

Perceptrón multicapa (MLP) con Keras para clasificar las tres especies de Iris: preparación de datos, construcción, compilación, entrenamiento, evaluación y predicción.

from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from tensorflow import keras
from tensorflow.keras import layers
from tensorflow.keras.utils import to_categorical

# Cargar y preparar los datos
iris = load_iris()
X = iris.data
y = iris.target

# Codificación one-hot de las etiquetas
y_encoded = to_categorical(y)

# División en entrenamiento y prueba
X_train, X_test, y_train, y_test = train_test_split(
    X, y_encoded, test_size=0.2, random_state=42)

# Escalado de las variables numéricas
scaler = StandardScaler()
X_train = scaler.fit_transform(X_train)
X_test = scaler.transform(X_test)

# Construcción del modelo
model = keras.Sequential([
    keras.Input(shape=(4,)),
    layers.Dense(8, activation="relu"),
    layers.Dense(6, activation="relu"),
    layers.Dense(3, activation="softmax"),  # 3 clases de salida
])

# Compilación del modelo
model.compile(optimizer="adam",
              loss="categorical_crossentropy",
              metrics=["accuracy"])

# Entrenamiento del modelo
history = model.fit(X_train, y_train, epochs=100, batch_size=8,
                    verbose=0, validation_split=0.1)

# Evaluación del modelo
loss, accuracy = model.evaluate(X_test, y_test, verbose=0)
print(f"Accuracy del modelo en el conjunto de prueba: {accuracy:.2f}")

# Predicción sobre nuevas observaciones
predicciones = model.predict(X_test[:5])
print("\nEjemplo de predicciones (probabilidades):\n", predicciones)

7.3.2Ejemplo de red convolucional

Pág. 106‑108Python

Red convolucional (CNN) que clasifica los dígitos manuscritos de MNIST.

from tensorflow import keras
from tensorflow.keras import layers
from tensorflow.keras.datasets import mnist
from tensorflow.keras.utils import to_categorical

# Cargar y preparar los datos
(X_train, y_train), (X_test, y_test) = mnist.load_data()

# Normalizar valores de píxeles (0-255) al rango [0, 1]
X_train = X_train.astype("float32") / 255.0
X_test = X_test.astype("float32") / 255.0

# Redimensionar para incluir el canal (1 para imágenes en escala de grises)
X_train = X_train.reshape(-1, 28, 28, 1)
X_test = X_test.reshape(-1, 28, 28, 1)

# Codificar las etiquetas como vectores one-hot
y_train = to_categorical(y_train, 10)
y_test = to_categorical(y_test, 10)

# Definir la arquitectura de la CNN
model = keras.Sequential([
    keras.Input(shape=(28, 28, 1)),
    layers.Conv2D(32, (3, 3), activation="relu"),
    layers.MaxPooling2D((2, 2)),
    layers.Conv2D(64, (3, 3), activation="relu"),
    layers.MaxPooling2D((2, 2)),
    layers.Flatten(),
    layers.Dense(64, activation="relu"),
    layers.Dense(10, activation="softmax"),
])

# Compilar el modelo
model.compile(optimizer="adam",
              loss="categorical_crossentropy",
              metrics=["accuracy"])

# Entrenar el modelo
history = model.fit(X_train, y_train, epochs=5, batch_size=64,
                    validation_split=0.1, verbose=1)

# Evaluar el rendimiento
test_loss, test_acc = model.evaluate(X_test, y_test, verbose=0)
print(f"\nExactitud en el conjunto de prueba: {test_acc:.3f}")

7.4.2Ejemplo de una RNN

Pág. 112‑113Python

Red LSTM para análisis de sentimientos de las reseñas de películas de IMDB (positiva o negativa).

from tensorflow import keras
from tensorflow.keras import layers
from tensorflow.keras.datasets import imdb
from tensorflow.keras.preprocessing.sequence import pad_sequences

# Cargar el dataset IMDB
num_palabras = 1000  # Se considerarán las 1000 palabras más frecuentes
(X_train, y_train), (X_test, y_test) = imdb.load_data(num_words=num_palabras)

print("Tamaño del conjunto de entrenamiento:", len(X_train))
print("Ejemplo de secuencia codificada:\n", X_train[0][:20])

# Preprocesamiento: normalización de la longitud de las secuencias
max_longitud = 100  # Todas las reseñas tendrán la misma longitud
X_train = pad_sequences(X_train, maxlen=max_longitud)
X_test = pad_sequences(X_test, maxlen=max_longitud)

# Construcción del modelo LSTM
model = keras.Sequential([
    keras.Input(shape=(max_longitud,)),
    layers.Embedding(input_dim=num_palabras, output_dim=64),
    layers.LSTM(128, return_sequences=False),
    layers.Dropout(0.2),
    layers.Dense(1, activation="sigmoid"),  # Salida binaria: positivo/negativo
])

# Compilación
model.compile(optimizer="adam", loss="binary_crossentropy",
              metrics=["accuracy"])

# Entrenamiento
history = model.fit(X_train, y_train, epochs=4, batch_size=128,
                    validation_split=0.2, verbose=1)

# Evaluación del modelo
loss, accuracy = model.evaluate(X_test, y_test, verbose=0)
print(f"\nExactitud del modelo en el conjunto de prueba: {accuracy:.3f}")

7.5.4Grandes modelos del lenguaje (LLMs)

Pág. 118Python

Genera texto con el modelo GPT‑2 de Hugging Face a partir de un prompt inicial.

# Requiere: pip install transformers torch
from transformers import GPT2LMHeadModel, GPT2Tokenizer

# Cargar modelo y tokenizador
tokenizer = GPT2Tokenizer.from_pretrained("openai-community/gpt2-medium")
model = GPT2LMHeadModel.from_pretrained("openai-community/gpt2-medium")

# Texto inicial (prompt)
prompt = "The most important thing in life is"
inputs = tokenizer(prompt, return_tensors="pt")

# Generar texto
outputs = model.generate(**inputs, max_new_tokens=30, do_sample=True,
                         temperature=0.8, top_p=0.9,
                         pad_token_id=tokenizer.eos_token_id)
texto_generado = tokenizer.decode(outputs[0], skip_special_tokens=True)

print("\nTexto generado:\n")
print(texto_generado)
